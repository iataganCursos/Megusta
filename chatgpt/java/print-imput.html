Campo Print e Campo Input

Arquivo: Exemplo.java
public class Exemplo {
    public static void x_print(String message) {
        System.out.print(message);
    }
    public static void x_println(String message) {
        System.out.println(message);
    }
    public static String x_input(String prompt) {
        System.out.print(prompt);
        java.util.Scanner scanner = new java.util.Scanner(System.in);
        return scanner.nextLine();
    }
    public static void main(String[] args) {
        x_print("Esta eh uma mensagem de exemplo.");

        x_println("Esta eh uma mensagem de exemplo.");

        String nome = x_input("Qual eh o seu nome? ");
        x_println("Olah, " + nome + "!");
    }
}
