// DigiteSeuNome.java

public class DigiteSeuNome {
    public static void main(String[] args) {
        String Texto = "";
        for (int x = 0; x <= args.length - 1; x++) {
            Texto += args[x] + " ";
        }
        System.out.println("Meu nome eh: "+Texto);
    }
}
