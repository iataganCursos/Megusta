Arquivo Externo

Arquivo: ExemploJava.java
import pastaexterna.arquivoexterno;

// ExemploJava.java

public class ExemploJava {
    public static void main(String[] args) {
        arquivoexterno f = new arquivoexterno();

        System.out.println(f.arquivo());
    }
}

Arquivo: arquivoexterno.java

Observação: deixe arquivoexterno.java dentro da pasta pastaexterna
package pastaexterna;

// arquivoesterno.java

public class arquivoexterno {

    public String arquivo() {

        String Texto;

        Texto = "";
        Texto += "Testando um arquivo externo. ";
        Texto += "Bom dia ";
        Texto += "Boa tarde e ";
        Texto += "Boa noite. ";
        Texto += "Fim do arquivo externo.";

        return Texto;

    }
}
Resultado:

C:\>javac ExemploJava.java

C:\>java ExemploJava
Testando um arquivo externo. Bom dia Boa tarde e Boa noite. Fim do arquivo externo.
