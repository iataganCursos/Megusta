Campo Print e Campo Input

Arquivo: Exemplo.py
def x_print(message: str):
    print(message, end="")  # Sem quebra de linha

def x_println(message: str):
    print(message)  # Com quebra de linha

def x_input(prompt: str) -> str:
    return input(prompt)

x_print("Esta é uma mensagem de exemplo.")

x_println("Esta é uma mensagem de exemplo.")

nome = x_input("Qual é o seu nome? ")
x_println("Olá, " + nome + "!")
